#!/usr/bin/env bash
# 08_share_intent - adb ACTION_SEND smoke test for share-sheet ingestion.
#
# Not a Maestro YAML flow: Android's ACTION_SEND cannot be delivered by
# `launchApp`, so this script drives the intent with adb and asserts on the
# resulting hierarchy.
#
# Usage: tools/maestro/flows_pending/08_share_intent.txt [text]
# Exit:  0 PASS, 1 FAIL, 2 environment error.
set -uo pipefail
ADB="${ADB:-$HOME/Android/Sdk/platform-tools/adb}"
DEVICE="${DEVICE:-emulator-5554}"
TEXT="${1:-The harbour town of Wick put itself to bed early, and the daughter of a lamplighter walked her round with a brass taper in her hand.}"

echo "== 08 share intent =="
if ! "$ADB" devices | awk -v d="$DEVICE" '$1==d && $2=="device"{f=1} END{exit(f?0:1)}'; then
  echo "FAIL: $DEVICE is not connected" >&2
  exit 2
fi

"$ADB" -s "$DEVICE" shell am force-stop app.novelspeak
echo "-- sending ACTION_SEND text/plain"
# The whole am command is quoted for the DEVICE shell: adb otherwise re-splits
# the extra value on spaces and "am" sees pkg=<first word>.
"$ADB" -s "$DEVICE" shell \
  "am start -W -a android.intent.action.SEND -t text/plain --es android.intent.extra.TEXT '$TEXT' -n app.novelspeak/.MainActivity" | tail -3
sleep 4

"$ADB" -s "$DEVICE" shell uiautomator dump /sdcard/share_ui.xml >/dev/null 2>&1
"$ADB" -s "$DEVICE" pull /sdcard/share_ui.xml /tmp/share_ui.xml >/dev/null 2>&1

python3 - <<'PY'
import sys, re, xml.etree.ElementTree as ET
try:
    root = ET.parse('/tmp/share_ui.xml').getroot()
except Exception as e:
    print(f"FAIL: could not read hierarchy: {e}")
    sys.exit(1)
texts, descs = [], []
for n in root.iter('node'):
    t = (n.get('text') or '').strip()
    d = (n.get('content-desc') or '').strip()
    if t: texts.append(t)
    if d: descs.append(d)
shared = any(t == 'Shared text' or re.fullmatch(r'Shared text · \d+ sentences', t) for t in texts)
sentence = any(re.fullmatch(r'Sentence \d+ of \d+', d) for d in descs)
player = 'Read' in texts
print(f"  'Shared text · N sentences' status: {shared}")
print(f"  sentence seek-bar state:            {sentence}")
print(f"  player Read visible:                {player}")
if shared and sentence and player:
    print("PASS: shared text was ingested into the reader")
    sys.exit(0)
print("FAIL: share intent did not reach the reader")
print("  texts:", texts[:12])
sys.exit(1)
PY
